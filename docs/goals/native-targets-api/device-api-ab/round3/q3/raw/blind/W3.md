import { cameraApp, type Photo } from '@markless/device';

export default function PhoneCameraButton() @{
	let photo = state<Photo | null>(null);

	<div>
		<button

				if (taken) photo = taken;
			}}

		</button>

		}
	</div>

import { state } from '@markless/core';
import { camera, cameraApp, photoLibrary, settings, sharesheet, type Photo } from '@markless/device';
import { base } from '@markless/ui';

export default function PhotoStudio() @{
	const cam = camera({ facing: 'back' });
	let last = state<Photo | null>(null);
	let message = state('');

	<main>
		<base.camera el={cam} />
		<button
			disabled={!cam.ready}
			onClick={async () => {
				try {
					last = await cam.takePhoto();
					message = '';
				} catch (error) {
					message = (error as Error).message;
				}
			}}
		>
			Take photo
		</button>
		<button
			onClick={async () => {
				const taken = await cameraApp.open();
				if (taken) last = taken;
			}}
		>
			Use phone camera
		</button>
		<button
			onClick={async () => {
				const chosen = await photoLibrary.open();
				if (chosen) last = chosen;
			}}
		>
			Choose a photo
		</button>
		@if (cam.access === 'denied') {
			<p>
				Camera access was refused.
				<button onClick={() => settings.open()}>Open Settings</button>
			</p>
		}
		@if (last) {
			<>
				<img src={last.url} width={last.width} height={last.height} alt="Last photo" />
				<button onClick={() => sharesheet.open({ url: last!.url })}>Share</button>
			</>
		}
		@if (message) {
			<p>{message}</p>
		}
	</main>
}

## Guesses
- The names `cameraApp` and `photoLibrary`, and that each has an `open()` method like `settings` and `sharesheet`.
- `open()` takes no arguments. It resolves to a `Photo`, or to `null` if the person cancels, so it does not throw on cancel.
- `@if` with no `@else` is allowed, and `photo` narrows to non-null inside it.
- Native `disabled` and `width`/`height` attributes on `<img>` work as usual.
- A plain `try`/`catch` inside a handler is fine for a failed `takePhoto()`, including a `NotAllowedError`.
- Inside the Share handler I used `last!` because I was not sure narrowing carries into the closure.

## Answers
A: `cameraApp`. Two other natural names: `systemCamera`, `cameraScreen`.
B: `photoLibrary`. Two other natural names: `gallery`, `photoPicker`.
C: `cameraApp.open()` gives back `Promise<Photo | null>`: the new photo, or `null` if the person cancels. `photoLibrary.open()` gives back `Promise<Photo | null>`: the chosen photo, or `null` if they cancel. With options for several photos, it might give back `Photo[]`.
