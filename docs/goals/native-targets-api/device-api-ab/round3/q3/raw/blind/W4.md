import { state } from '@markless/core';
import { systemCamera, type Photo } from '@markless/device';

export default function UsePhoneCamera() @{
	let photo = state<Photo | null>(null);

	<div>
		<button
			onClick={async () => {
				const taken = await systemCamera.open();
				if (taken) photo = taken;
			}}
		>
			Use phone camera
		</button>
		@if (photo) {
			<img src={photo.url} width={photo.width} height={photo.height} alt="Photo taken" />
		}
	</div>
}

import { state } from '@markless/core';
import { camera, photoLibrary, settings, sharesheet, systemCamera, type Photo } from '@markless/device';
import { base } from '@markless/ui';

export default function PhotoStudio() @{
	const cam = camera({ facing: 'back' });
	let last = state<Photo | null>(null);
	let problem = state('');

	<main>
		<base.camera el={cam} />
		<button
			disabled={cam.access === 'denied'}
			onClick={async () => {
				try {
					last = await cam.takePhoto();
					problem = '';
				} catch (error) {
					problem = (error as Error).name === 'NotAllowedError' ? 'Camera access was refused.' : (error as Error).message;
				}
			}}
		>
			Shutter
		</button>
		<button
			onClick={async () => {
				const taken = await systemCamera.open();
				if (taken) last = taken;
			}}
		>
			Use phone camera
		</button>
		<button
			onClick={async () => {
				const chosen = await photoLibrary.open();
				if (chosen) last = chosen;
			}}
		>
			Choose a photo
		</button>
		@if (cam.access === 'denied') {
			<button onClick={() => settings.open()}>Open Settings</button>
		}
		@if (problem) {
			<p>{problem}</p>
		}
		@if (last) {
			<>
				<img src={last.url} width={last.width} height={last.height} alt="Last photo" />
				<button
					onClick={async () => {
						if (last) await sharesheet.open({ url: last.url });
					}}
				>
					Share
				</button>
			</>
		}
	</main>
}
