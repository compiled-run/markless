import { cameraScreen, type Photo } from '@markless/device';

export default function PhoneCameraButton() @{
	let photo = state<Photo | null>(null);

	<div>
		<button

				if (taken) photo = taken;
			}}
		>

		</button>
		@if (photo) {

		}
	</div>
}

// file: PhotoStudio.tsrx
import { state } from '@markless/core';
import {
	camera,
	cameraScreen,
	photoLibrary,
	settings,
	sharesheet,
	type Photo,
} from '@markless/device';
import { base } from '@markless/ui';

export default function PhotoStudio() @{
	const cam = camera({ facing: 'back' });
	let last = state<Photo | null>(null);
	let problem = state('');

	<main>
		<base.camera el={cam} />
		<button
			disabled={!cam.ready}
			onClick={async () => {
				try {
					last = await cam.takePhoto();
					problem = '';
				} catch (error) {
					problem = (error as Error).message;
				}
			}}
		>
			Take photo
		</button>
		<button
			onClick={async () => {
				const taken = await cameraScreen.open();
				if (taken) last = taken;
			}}
		>
			Use phone camera
		</button>
		<button
			onClick={async () => {
				const chosen = await photoLibrary.open();
				if (chosen) last = chosen;
			}}
		>
			Choose a photo
		</button>
		@if (cam.access === 'denied') {
			<button onClick={() => settings.open()}>Open Settings</button>
		}
		@if (last) {
			<>
				<img src={last.url} width={last.width} height={last.height} alt="Last photo" />
				<button onClick={() => sharesheet.open({ url: last?.url })}>Share</button>
			</>
		}
		@if (problem) {
			<p>{problem}</p>
		}
	</main>
}

## Guesses
- The names `cameraScreen` and `photoLibrary`. I also guessed that each one has an `open()` method that returns `Promise<Photo | null>`.
- I assumed `null` (or `undefined`) means the person cancelled. The `if (taken)` guard handles either. The screens might reject on cancel instead.
- I assumed a handler can assign state after an `await`, and that `@if (last)` narrows `last` inside the block. I used `last?.url` in the Share handler to be safe.
- I disabled the shutter while `!cam.ready`. That isn't strictly needed, because `takePhoto` waits until the camera is ready.
- I put the `Photo` `url` in a plain `<img src>`, as the sheet says it works there.

## Answers
A: `cameraScreen`. Two other natural names: `systemCamera` and `cameraApp`.
B: `photoLibrary`. Two other natural names: `photoPicker` and `gallery`.
C: Both `cameraScreen.open()` and `photoLibrary.open()` return `Promise<Photo | null>`. They resolve with the `Photo` taken or chosen (it has `url`, `width` and `height`), or with `null` if the person cancels. If permission is refused, they reject with an Error whose name is `'NotAllowedError'`.
