import { state } from '@markless/core';
import { cameraApp, type Photo } from '@markless/device';

export default function CameraButton() @{
	let photo = state<Photo | null>(null);

	<section>
		<button
			onClick={async () => {
				const taken = await cameraApp.open();
				if (taken) photo = taken;
			}}
		>
			Use phone camera
		</button>
		@if (photo) {
			<img src={photo.url} alt="Photo taken" />
		}
	</section>
}

import { state } from '@markless/core';
import { camera, cameraApp, photoLibrary, settings, sharesheet, type Photo } from '@markless/device';
import { base } from '@markless/ui';

export default function PhotoStudio() @{
	const cam = camera({ facing: 'back' });
	let photo = state<Photo | null>(null);

	<main>
		<base.camera el={cam} />
		<button disabled={!cam.ready} onClick={async () => (photo = await cam.takePhoto())}>
			Shutter
		</button>
		@if (cam.access === 'denied') {
			<button onClick={() => settings.open()}>Open Settings</button>
		}
		<button
			onClick={async () => {
				const taken = await cameraApp.open();
				if (taken) photo = taken;
			}}
		>
			Use phone camera
		</button>
		<button
			onClick={async () => {
				const chosen = await photoLibrary.open();
				if (chosen) photo = chosen;
			}}
		>
			Choose a photo
		</button>
		@if (photo) {
			<>
				<img src={photo.url} alt="Latest photo" />
				<button
					onClick={async () => {
						if (photo) await sharesheet.open({ url: photo.url });
					}}
				>
					Share
				</button>
			</>
		}
	</main>
}
