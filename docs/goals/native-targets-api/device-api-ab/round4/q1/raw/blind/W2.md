```tsx
// file: pages/orders/[id].tsrx
import { computed, storage } from '@markless/core';
import { back, navigate, type PageProps } from '@markless/router';
import { platform } from '@markless/device';
import { modal } from '@markless/ui';
import { deleteOrder, getOrder } from '../../api.ts';

export default function OrderPage({ params }: PageProps<{ id: string }>) @{
	let ordersVersion = storage('ordersVersion', '0');
	const order = computed(async ({ signal }) => await getOrder(params.id, signal));

	const confirmDelete = async () => {
		await deleteOrder(params.id);
		ordersVersion = String(Number(ordersVersion) + 1);
		if (platform === 'web') {
			await navigate('/orders', { replace: true });
		} else {
			back();
		}
	};

	<main>
		@try {
			<section>
				<h1>Order {order.id}</h1>
				<p>{order.placedAt} · {order.total}</p>
			</section>
		} @pending {
			<p>Loading…</p>
		} @catch (error, reset) {
			<p>
				{error.message}
				<button onClick={reset}>Retry</button>
			</p>
		}
		<modal.root alert>
			<modal.trigger>Delete</modal.trigger>
			<modal.backdrop>
				<modal.content>
					<modal.title>Delete this order?</modal.title>
					<button onClick={confirmDelete}>Delete</button>
					<modal.close>Cancel</modal.close>
				</modal.content>
			</modal.backdrop>
		</modal.root>
	</main>
}
```

```tsx
// file: pages/checkout.tsrx
import { computed, storage } from '@markless/core';
import { Link } from '@markless/router';
import { getAddresses } from '../api.ts';

export default function Checkout() @{
	let deliverTo = storage('deliverTo', '');
	const address = computed(async ({ signal }) => {
		const id = deliverTo;
		const list = await getAddresses(signal);
		return list.find((a) => a.id === id) ?? list[0] ?? null;
	});
	<main>
		<h1>Checkout</h1>
		<p>
			Deliver to:{' '}
			@try {
				<span>{address ? `${address.label}, ${address.line1}, ${address.city}` : 'No address yet'}</span>
			} @pending {
				<span>Loading…</span>
			} @catch (error, reset) {
				<button onClick={reset}>{error.message}</button>
			}
			<Link href="/checkout/address">Change</Link>
		</p>
	</main>
}
```

```tsx
// file: pages/checkout/address.tsrx
import { computed, state, storage } from '@markless/core';
import { back, type Presentation } from '@markless/router';
import { addAddress, getAddresses } from '../../api.ts';

export const presentation: Presentation = 'sheet';

export default function ChooseAddress() @{
	let deliverTo = storage('deliverTo', '');
	let adding = state(false);
	let label = state('');
	let line1 = state('');
	let city = state('');
	let saveError = state('');
	const addresses = computed(async ({ signal }) => await getAddresses(signal));

	const save = async (event: SubmitEvent) => {
		event.preventDefault();
		try {
			const created = await addAddress({ label, line1, city });
			deliverTo = created.id;
			back();
		} catch (error) {
			saveError = (error as Error).message;
		}
	};

	<section aria-label="Deliver to">
		@if (adding) {
			<form onSubmit={save}>
				<h1>New address</h1>
				<input placeholder="Label" value={label} onInput={(e) => (label = e.currentTarget.value)} />
				<input placeholder="Street" value={line1} onInput={(e) => (line1 = e.currentTarget.value)} />
				<input placeholder="City" value={city} onInput={(e) => (city = e.currentTarget.value)} />
				<p>{saveError}</p>
				<button type="button" onClick={() => (adding = false)}>Back to list</button>
				<button type="submit">Save</button>
			</form>
		} @else {
			<div>
				<h1>Deliver to</h1>
				@try {
					<ul>
						@for (const address of addresses; key address.id) {
							<li>
								<button onClick={() => { deliverTo = address.id; back(); }}>
									{address.label}: {address.line1}, {address.city}
								</button>
							</li>
						}
					</ul>
				} @pending {
					<p>Loading…</p>
				} @catch (error, reset) {
					<p>
						{error.message}
						<button onClick={reset}>Retry</button>
					</p>
				}
				<button onClick={() => (adding = true)}>Add new address</button>
			</div>
		}
	</section>
}
```

## Guesses

- Whether a page file may export something other than its default component (`presentation`), and whether the router reads it on every platform.
- That swiping down on a sheet is the system Back, so it pops the sheet the same way `back()` does.
- That reading `storage` inside an async `computed` (before the `await`) makes it run again when the value changes.
- That async event handlers, and `try`/`catch` inside them, are allowed.
- That a normal `<button>` inside `modal.content` is fine, and that leaving the page tears the dialog down.
- That on a phone `/orders` always sits under `/orders/[id]`. That holds when you arrive from Orders or from a deep link, because the router builds the leading screens.
- That `{' '}` and a `@try` mixed with text inside one `<p>` count as valid children.
- That when the notification is tapped while the app is running, the router rebuilds the stack as `/` then `/cart`, the same as a cold start.

## Website

1. Cart: `/cart` is an ordinary full page. The Cart links load it, and the browser's Back returns to the page before it.
2. Delete order: the same confirm dialog. Delete then replaces the page with a freshly loaded `/orders`.
3. Change address: `/checkout/address` is a full page. Picking an address saves it in the cookie and goes back, and checkout loads again showing the choice.
4. Add new address: the form is a second step on that same page. Saving goes back to checkout with the new address picked.
5. Notification: there is no push on the website. A link to `/cart` just opens the cart page.

## Answers

A: A page becomes a sheet by exporting `presentation = 'sheet'` from its page file. Close it with `back()` or a swipe down, and pass results out through `storage` before calling `back()`, because code after `back()` does not run.

B: The decision could have lived on the link, on the page that opens the sheet, in app config, or on the page that is shown. I put it on the page that is shown. That way every way of reaching the page gets the same result, including a Link from any of the three pages, `navigate()`, and a deep link or notification that has no "opener" at all. It also sits next to the page's own code instead of in a separate route table.

C: Picking an address assigns `storage('deliverTo')` and then calls `back()`. On a phone, checkout is still on the stack, and the storage write updates it at once; its computed reads `deliverTo`, so it runs again. On the website the value is saved in a cookie, and checkout loads again on Back and reads the same key.

D: The app opens with Home as the base screen and the cart sheet slid up over it. Back or a swipe down closes the sheet and leaves Home, and Back again leaves the app.
