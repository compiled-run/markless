```ts
// Module: @markless/router (addition)
// How a page appears when something goes to it on iPhone and Android. Ignored on the website.
export type Presentation = 'screen' | 'sheet';
// Optional named export from a page file. 'sheet' makes the page a sheet that slides up over the screen below it. That screen stays visible behind it. The system Back, which includes a swipe down on the sheet, pops it like any other screen. The default is 'screen'.
export declare const presentation: Presentation; // written in a page file as: export const presentation = 'sheet';
```

```tsrx
// file: pages/orders/[id].tsrx
import { computed, state, storage } from '@markless/core';
import { back, navigate, type PageProps } from '@markless/router';
import { platform } from '@markless/device';
import { modal } from '@markless/ui';
import { deleteOrder, getOrder } from '../../api.ts';

export default function OrderPage({ params }: PageProps<{ id: string }>) @{
	let ordersChanged = storage('ordersChanged', '');
	let failure = state('');
	const order = computed(async ({ signal }) => await getOrder(params.id, signal));

	async function remove() {
		try {
			await deleteOrder(params.id);
		} catch (error) {
			failure = (error as Error).message;
			return;
		}
		ordersChanged = String(Date.now());
		if (platform === 'web') {
			void navigate('/orders', { replace: true });
		} else {
			back();
		}
	}

	<main>
		@try {
			<h1>Order {order.id} — {order.total}</h1>
		} @pending {
			<p>Loading…</p>
		} @catch (error, reset) {
			<button onClick={reset}>{error.message}</button>
		}
		<modal.root alert>
			<modal.trigger>Delete</modal.trigger>
			<modal.backdrop>
				<modal.content>
					<modal.title>Delete this order?</modal.title>
					@if (failure !== '') {
						<modal.description>{failure}</modal.description>
					}
					<button onClick={remove}>Delete</button>
					<modal.close>Cancel</modal.close>
				</modal.content>
			</modal.backdrop>
		</modal.root>
	</main>
}
```

```tsrx
import { computed, storage } from '@markless/core';
import { navigate } from '@markless/router';
import { getAddresses, type Address } from '../api.ts';

function describe(addresses: Address[], id: string): string {
	const found = addresses.find((address) => address.id === id);
	return found ? `${found.label}, ${found.line1}, ${found.city}` : 'No address chosen';
}

export default function Checkout() @{
	let deliveryAddress = storage('deliveryAddress', '');
	const addresses = computed(async ({ signal }) => {
		deliveryAddress;
		return await getAddresses(signal);

				<button onClick={reset}>{error.message}</button>
```

```tsrx
// file: pages/checkout/address.tsrx
import { computed, state, storage } from '@markless/core';
import { back, navigate } from '@markless/router';
import { platform } from '@markless/device';
import { addAddress, getAddresses } from '../../api.ts';

export const presentation = 'sheet';

export default function ChooseAddress() @{
	let deliveryAddress = storage('deliveryAddress', '');
	let step = state<'list' | 'new'>('list');
	let label = state('');
	let line1 = state('');
	let city = state('');
	let failure = state('');
	const addresses = computed(async ({ signal }) => await getAddresses(signal));

	function finish(id: string) {
		deliveryAddress = id;
		if (platform === 'web') {
			void navigate('/checkout', { replace: true });
		} else {
			back();
		}
	}

	async function save() {
		try {
			const created = await addAddress({ label, line1, city });
			finish(created.id);
		} catch (error) {
			failure = (error as Error).message;
		}
	}

	<section>
		@if (step === 'list') {
			<div>
				<h1>Deliver to</h1>
				@try {
					<ul>
						@for (const address of addresses; key address.id) {
							<li>
								<button onClick={() => finish(address.id)}>
									{address.label} — {address.line1}, {address.city}
								</button>
							</li>
						}
					</ul>
				} @pending {
					<p>Loading…</p>
				} @catch (error, reset) {
					<p>
						{error.message}
						<button onClick={reset}>Retry</button>
					</p>
				}
				<button onClick={() => (step = 'new')}>Add new address</button>
			</div>
		} @else {
			<form onSubmit={(event) => { event.preventDefault(); void save(); }}>
				<h1>New address</h1>
				<input placeholder="Label" value={label} onInput={(event) => (label = event.currentTarget.value)} />
				<input placeholder="Street" value={line1} onInput={(event) => (line1 = event.currentTarget.value)} />
				<input placeholder="City" value={city} onInput={(event) => (city = event.currentTarget.value)} />
				@if (failure !== '') {
					<p>{failure}</p>
				}
				<button type="button" onClick={() => (step = 'list')}>Back</button>
				<button type="submit">Save</button>
			</form>
		}
	</section>
}
```

## Guesses

- The sheet is decided by a named `presentation` export from the page file, read when the app is built. A sheet's screen keeps its own state and is popped by Back, which includes the swipe down.
- Opening the app at `/cart` from outside it builds `/` as a normal screen, then `/cart` as a sheet over it.
- A `<Link replace>` inside a sheet that goes to a normal page (Cart to Checkout) replaces the sheet with a full screen.
- On the website, `back()` may show the cached older page, so I use `navigate(..., { replace: true })` there.
- Ordinary functions (`finish`, `save`, `remove`) can be declared in the body, and assigning state or storage inside them works.
- An async handler can assign storage and then call `back()` after `await`.
- A storage assignment made before `back()` reaches the screen underneath. That screen's computeds that read the key run again.
- I used a time stamp in storage (`ordersChanged`) so the Orders screen kept on the stack fetches its list again.
- `{' '}` and text plus a `@try` beside each other inside `<p>` are allowed.
- `modal.root` with only `alert` and no `open` starts closed. `modal.description` can hold an error message.

## Website

1. Cart: `/cart` is an ordinary full page that loads in place. There is no Close button; the visitor uses the browser's Back or the Checkout link.
2. Delete order: the same dialog appears. After Delete, the Orders page loads in place of the order page, so the browser's Back does not return to the deleted order.
3. Address choice: "Change" loads `/checkout/address` as a full page. Picking an address saves it to the cookie-backed storage, then loads `/checkout` in place, and that page shows the choice.
4. New address: the same two steps happen inside the `/checkout/address` page. Saving goes back to `/checkout` with the new address chosen.
5. Notification: website visitors get no push notification. Anyone who opens `/cart` directly gets the full cart page.

## Answers

A: Add `export const presentation = 'sheet'` to a page file and every visit to that page on iPhone and Android opens it as a sheet over the current screen, which Back or a swipe down closes, while the website shows it as a normal page.

B: The decision could have lived on each link, in the page that opens the sheet (for example rendering an inline sheet component), in app config, or in the page being shown. I picked the page being shown, because then all three Cart links, `navigate()` calls and the notification's deep link agree with no extra code, and the sheet is still a route with its own URL.

C: The address sheet writes the chosen id to `storage('deliveryAddress')`. On a phone, that assignment updates the checkout screen kept underneath and `back()` drops the sheet, so checkout shows the choice right away. On the website the storage is a cookie, so the `/checkout` page that loads after the choice reads the new id when it renders.

D: The notification carries `/cart`, so the app builds Home and opens the cart sheet over it. Back, or a swipe down on the sheet, closes the cart and leaves the person on Home; a further Back leaves the app.
