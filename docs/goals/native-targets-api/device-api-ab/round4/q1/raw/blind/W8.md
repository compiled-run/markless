```tsrx
// file: pages/orders/[id].tsrx
import { computed, state, storage } from '@markless/core';
import { back, navigate, type PageProps } from '@markless/router';
import { platform } from '@markless/device';
import { modal } from '@markless/ui';
import { deleteOrder, getOrder } from '../../api.ts';

export default function OrderPage({ params }: PageProps<{ id: string }>) @{
	let ordersVersion = storage('ordersVersion', '0');
	let deleteError = state('');
	const order = computed(async ({ signal }) => await getOrder(params.id, signal));
	const confirmDelete = async () => {
		try {
			await deleteOrder(params.id);
		} catch (error) {
			deleteError = (error as Error).message;
			return;
		}
		ordersVersion = String(Number(ordersVersion) + 1);
		if (platform === 'web') {
			await navigate('/orders', { replace: true });
		} else {
			back();
		}
	};
	<main>
		@try {
			<h1>Order {order.id}: {order.total}</h1>
		} @pending {
			<p>Loading…</p>
		} @catch (error, reset) {
			<button onClick={reset}>{error.message}</button>
		}
		<modal.root alert>
			<modal.trigger>Delete</modal.trigger>
			<modal.backdrop>
				<modal.content>
					<modal.title>Delete this order?</modal.title>
					@if (deleteError !== '') {
						<modal.description>{deleteError}</modal.description>
					}
					<button onClick={confirmDelete}>Delete</button>
					<modal.close>Cancel</modal.close>
				</modal.content>
			</modal.backdrop>
		</modal.root>
	</main>
}
```

```tsrx
// file: pages/checkout/address.tsrx
import { computed, state, storage } from '@markless/core';
import { back, navigate } from '@markless/router';
import { platform } from '@markless/device';
import { addAddress, getAddresses } from '../../api.ts';

export const presentation = 'sheet';

export default function AddressSheet() @{
	let addressId = storage('deliveryAddressId', '');
	let step = state<'list' | 'form'>('list');
	let label = state('');
	let line1 = state('');
	let city = state('');
	let saveError = state('');
	const addresses = computed(async ({ signal }) => await getAddresses(signal));
	const close = () => {
		if (platform === 'web') {
			navigate('/checkout', { replace: true });
		} else {
			back();
		}
	};
	<section>
		<h2>Deliver to</h2>
		@if (step === 'list') {
			<>
				@try {
					<ul>
						@for (const address of addresses; key address.id) {
							<li>
								<button onClick={() => { addressId = address.id; close(); }}>
									{address.label}: {address.line1}, {address.city}
								</button>
							</li>
						}
					</ul>
				} @pending {
					<p>Loading…</p>
				} @catch (error, reset) {
					<button onClick={reset}>{error.message}</button>
				}
				<button onClick={() => (step = 'form')}>Add new address</button>
			</>
		} @else {
			<form
				onSubmit={async (event) => {
					event.preventDefault();
					try {
						const added = await addAddress({ label, line1, city });
						addressId = added.id;
					} catch (error) {
						saveError = (error as Error).message;
						return;
					}
					close();
				}}
			>
				<input placeholder="Label" value={label} onInput={(event) => (label = event.currentTarget.value)} />
				<input placeholder="Street" value={line1} onInput={(event) => (line1 = event.currentTarget.value)} />
				<input placeholder="City" value={city} onInput={(event) => (city = event.currentTarget.value)} />
				@if (saveError !== '') {
					<p>{saveError}</p>
				}
				<button type="submit">Save</button>
				<button type="button" onClick={() => (step = 'list')}>Back to list</button>
			</form>
		}
	</section>
}
```

Task 5: the notification carries `path: '/cart'`. The app needs no extra code, because `pages/cart.tsrx` is already a sheet route. The server sends `{ title: 'Your cart is waiting', path: '/cart' }`.

Confidence: 3/5

## Guesses

- The router reads the page file's `presentation` export at build time, much as it reads the default export.
- Reading a `storage` value inside an async computed is a dependency, the same as reading a `state`, so the computed runs again when that value is assigned.
- Only `storage` crosses screens, so I pass results back by storage key: `deliveryAddressId` for the chosen address, and `ordersVersion` to make the Orders list fetch again.
- An `async` handler can `await` before it calls `navigate`/`back`. A `return` placed before them does what you would expect.
- Plain `const` arrow functions declared in the body can be used as handlers. `try`/`catch` works inside them.
- A text node mixing several `{}` expressions (`{line.name} × {line.quantity}`) counts as content of one element.
- `<modal.description>` can sit under an `@if` inside the content.
- A deep link to `/checkout/address` builds `/`, `/checkout`, then the sheet. That is why `back()` is safe on phones.
- On the website, deep links into a sheet page are possible, so I use `navigate(..., { replace: true })` there instead of `back()`.

## Website

1. `/cart` is an ordinary full page. The browser's Back goes to the page the visitor came from.
2. The same "Delete this order?" alert dialog appears. After Delete, the browser loads the Orders page in place of the order page, fetched fresh.
3. "Change" loads `/checkout/address` as a full page. Picking an address saves it in the cookie and loads `/checkout`, which shows the new choice.
4. The same two steps happen on that page. Saving adds the address, picks it and loads `/checkout`.
5. The website gets no push notification. Visitors reach the cart through the "Cart" links at `/cart`.

## Answers

A: A page file that has `export const presentation = 'sheet'` opens as a bottom sheet over the current screen on iPhone and Android, and as a normal page on the website; everything else (Link, navigate, back, swipe down = Back, deep links, its own fresh state) works as for any screen.

B: The other places it could live:

- **The link**, for example `<Link sheet>`. The same route could then appear in different ways, and a deep link or notification would have no link to ask.
- **The page that opens it**, for example a `<sheet>` component holding the cart. You would have to copy it into three pages, and a notification could not reach it.
- **App config.** This splits the decision away from the page it describes.

I picked **the page that is shown**. It is a property of the route, so links, `navigate`, deep links and notifications all agree with no extra code.

C: On a phone, the sheet assigns `storage('deliveryAddressId')` and calls `back()`. Storage updates every screen at once, so the checkout screen kept underneath sees the new id, its computed runs again, and it shows the choice when the sheet goes away. On the website, the same storage key is a cookie. The sheet page sets it and navigates to `/checkout`, and the server renders checkout with the new cookie.

D: The app builds `/` (Home), then opens the cart sheet over it, so the user sees the cart sheet with Home visible behind it. Back, or a swipe down, closes the sheet and leaves them on Home. A second Back leaves the app.
