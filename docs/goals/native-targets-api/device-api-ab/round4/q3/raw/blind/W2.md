```tsrx
// file: pages/media.tsrx
import { state } from '@markless/core';
import { camera, CameraView, PdfView, type Photo } from '@markless/device';
import { tabs, base } from '@markless/ui';

function CameraPanel() @{
	const cam = camera({ facing: 'back' });
	let photo = state<Photo | null>(null);

	<div class="panel">
		<CameraView el={cam} class="fill" />
		<div class="controls">
			<button onClick={async () => (photo = await cam.takePhoto())}>Take photo</button>
			<base.button pressed={cam.facing === 'front'} onChange={() => cam.flip()}>
				Flip camera
			</base.button>
		</div>
		@if (photo) {
			<img class="fill" src={photo.url} alt="Last photo" />
		}
	</div>
}

function DocumentPanel() @{
	let page = state(1);
	let pages = state(0);

	<div class="panel">
		<PdfView src="/guide.pdf" page={page} onLoad={(info) => (pages = info.pages)} class="fill" />
		<p>Page {page} of {pages}</p>
		<div class="controls">
			<button disabled={page <= 1} onClick={() => page--}>Previous</button>
			<button disabled={pages === 0 || page >= pages} onClick={() => page++}>Next</button>
		</div>
	</div>
}

export default function MediaPage() @{
	let tab = state('camera');

	<main>
		<style>{`
			.panel { display: flex; flex-direction: column; gap: 0.5rem; }
			.fill { display: block; width: 100%; }
			.controls { display: flex; gap: 0.5rem; }
		`}</style>
		<tabs.root value={tab} onChange={(value) => (tab = value)}>
			<tabs.list>
				<tabs.trigger value="camera">Camera</tabs.trigger>
				<tabs.trigger value="document">Document</tabs.trigger>
			</tabs.list>
			<tabs.content value="camera">
				<CameraPanel />
			</tabs.content>
			<tabs.content value="document">
				<DocumentPanel />
			</tabs.content>
		</tabs.root>
	</main>
}
```

## Guesses

- I guessed the element names `CameraView` and `PdfView`, and that they are imported from `@markless/device` as components.
- That `Photo` is exported as a type from `@markless/device`.
- That `base.button` is used as `<base.button>` in markup. I made it a toggle with `pressed={cam.facing === 'front'}` because the sheet gives a plain button no `onClick`, and `onChange` fires only on a toggle. I also guessed that the toggle follows the live `facing` value after `flip()`.
- That `tabs.root` accepts a state value for `value` and that `onChange` writing it back is the intended pattern.
- That an `async` arrow handler which assigns the awaited result to state is allowed.
- That a `<style>` element containing a template string works inside a `.tsrx` body, and that `class` (not `className`) is the attribute name.
- That `@if (photo)` narrows `photo` from `null`, so `photo.url` type-checks.
- That the camera keeps running while its tab panel is hidden, because hidden content stays in the page. I did not stop it.
- That `page` and `pages` render as text inside `<p>` next to static text.

## Answers

A: From `@markless/device`, next to `camera()`. The camera element only shows a `Camera` made by that module, and both elements are drawn by the device rather than being HTML or styleable `@markless/ui` parts. So the device package is the natural owner.

B: `<CameraView el={cam} />`. I would also find `<device.camera el={cam} />` and `<CameraPreview el={cam} />` natural.

C: Neither, most likely. The capitalized `CameraView` reads as an imported component, not a lowercase HTML tag. It is not written in the `family.part` form (`tabs.trigger`, `base.button`) that `@markless/ui` parts use. A reader could still take it for an ordinary user-written component, because nothing in the name says it is a native device view.
