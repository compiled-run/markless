declare global {
	namespace JSX {
		interface IntrinsicElements {
			camera: { el: import('@markless/device').Camera; class?: string };
			pdf: { src: string; page?: number; onLoad?: (info: { pages: number }) => void; class?: string };
		}
	}

```tsrx
// file: pages/media.tsrx
import { state } from '@markless/core';
import { camera, type Photo } from '@markless/device';
import { tabs, base } from '@markless/ui';

function CameraPanel() @{
	const cam = camera({ facing: 'back' });
	let photo = state<Photo | null>(null);
	let failure = state('');

	<div class="panel">
		<camera el={cam} class="fill" />
		<div class="controls">
			<button
				disabled={!cam.ready}
				onClick={async () => {
					try {
						photo = await cam.takePhoto();
						failure = '';
					} catch (error) {
						failure = (error as Error).message;
					}
				}}
			>
				Take photo
			</button>
			<base.button pressed={cam.facing === 'front'} onChange={() => cam.flip()}>
				Flip camera
			</base.button>
		</div>
		@if (cam.access === 'denied') {
			<p>Camera access was denied.</p>
		}
		@if (failure) {
			<p>{failure}</p>
		}
		@if (photo) {
			<img class="fill" src={photo.url} width={photo.width} height={photo.height} alt="Last photo" />
		}
	</div>
}

function DocumentPanel() @{
	let page = state(1);
	let pages = state(0);

	<div class="panel">
		<pdf src="/guide.pdf" page={page} onLoad={(info) => (pages = info.pages)} class="fill" />
		<div class="controls">
			<button disabled={page <= 1} onClick={() => page--}>Previous</button>
			<span>Page {page} of {pages}</span>
			<button disabled={pages === 0 || page >= pages} onClick={() => page++}>Next</button>
		</div>
	</div>
}

export default function Media() @{
	let tab = state('camera');

	<main>
		<style>
			.panel { display: flex; flex-direction: column; gap: 0.5rem; }
			.fill { display: block; width: 100%; }
			.controls { display: flex; gap: 0.5rem; align-items: center; }
		</style>
		<tabs.root value={tab} onChange={(value) => (tab = value)}>
			<tabs.list>
				<tabs.trigger value="camera">Camera</tabs.trigger>
				<tabs.trigger value="document">Document</tabs.trigger>
			</tabs.list>
			<tabs.content value="camera">
				<CameraPanel />
			</tabs.content>
			<tabs.content value="document">
				<DocumentPanel />
			</tabs.content>
		</tabs.root>
	</main>
}
```

## Guesses

- I guessed the tag names `<camera>` and `<pdf>`, and that they are intrinsic tags that need no import.
- I guessed that `@markless/ui` parts are written as `<tabs.root>` and `<base.button>` with dotted member tags.
- `base.button` has no `onClick` prop. So the flip button is a toggle: its `pressed` state is `cam.facing === 'front'`, and `onChange` calls `cam.flip()`. I assumed that `pressed` follows `cam.facing`, because the camera's fields are live.
- I guessed that a controlled `tabs.root` needs `onChange` to write `value` back to state. I also guessed that state can be a plain string here.
- I guessed that `async` click handlers are allowed and that assigning state after `await` updates the view.
- I guessed that a `<style>` block inside the component is supported and that its rules apply to these elements.
- I guessed that a component's body still runs, and its camera is held, while its tab panel is hidden, because the content part "stays in the page". So the camera may keep running while the Document tab shows. I also assumed `ready` handles a hidden preview.
- I guessed that `width` and `height` on `<img>` accept numbers, and that `catch (error)` needs a cast to read `.message`.
- I guessed that `@if` takes a parenthesized condition and narrows `photo` to non-null inside the block.

## Website

Camera preview: the browser asks permission to use the camera, then shows the live webcam picture full width. This is presumably a `<video>` element.

PDF page: the browser's built-in PDF viewer with its own toolbar, drawn as a full-width box that opens at the current page.

## Answers

A: They come from nowhere: I treated both as built-in intrinsic tags that the Markless compiler knows about. The sheet says each is "one box with no parts and no behaviour" and lists them beside HTML ("no HTML equivalent"). That reads like a platform primitive, the same kind of thing as `<div>`, not a component exported from `@markless/device` or `@markless/ui`. `@markless/device` was the next most likely home for both.

B: `<camera el={cam} />`. Two other spellings I would find natural are `<Camera.View el={cam} />`, imported from `@markless/device`, and `<CameraPreview el={cam} />`.

C: A reader could take it for an HTML element. It is lowercase, has no import and no dot, and looks just like `<video>` or `<canvas>`. Nothing in the code shows it is a Markless or native element. It would not be taken for a `@markless/ui` part, because those are always written dotted (`tabs.root`, `base.button`).
