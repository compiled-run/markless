// Module: @markless/device (guessed additions)
import type { Camera } from '@markless/device';

// Shows the live preview of a camera made with camera().
export declare function CameraPreview(props: { el: Camera; class?: string }): Element;

// Shows one page of a PDF. `page` counts from 1; `onLoad` is called once with the page count.
export declare function PdfPage(props: {

	page?: number;
	onLoad?: (info: { pages: number }) => void;

```tsx
// file: pages/media.tsrx
import { state } from '@markless/core';
import { camera, CameraPreview, PdfPage } from '@markless/device';
import type { Photo } from '@markless/device';
import { base, tabs } from '@markless/ui';

function CameraPanel() @{
	const cam = camera({ facing: 'back' });
	let photo = state<Photo | null>(null);

	<div class="camera-panel">
		<CameraPreview el={cam} class="fill-width" />
		<div class="camera-controls">
			<button onClick={async () => (photo = await cam.takePhoto())}>Take photo</button>
			<base.button pressed={cam.facing === 'front'} onChange={() => cam.flip()}>
				Flip camera
			</base.button>
		</div>
		@if (photo) {
			<img class="fill-width" src={photo.url} alt="Last photo taken" />
		}
	</div>
}

function DocumentPanel() @{
	let page = state(1);
	let pages = state<number | null>(null);

	<div class="document-panel">
		<PdfPage
			src="/guide.pdf"
			page={page}
			onLoad={(info) => (pages = info.pages)}
			class="fill-width"
		/>
		<p>
			Page {page} of {pages ?? '…'}
		</p>
		<div class="document-controls">
			<button disabled={page <= 1} onClick={() => page--}>Previous</button>
			<button disabled={pages === null || page >= pages} onClick={() => page++}>Next</button>
		</div>
	</div>
}

export default function Media() @{
	let tab = state('camera');

	<main>
		<style>{`
			.fill-width { display: block; width: 100%; }
			.camera-panel .fill-width { aspect-ratio: 3 / 4; }
			.document-panel .fill-width { aspect-ratio: 1 / 1.414; }
		`}</style>
		<tabs.root value={tab} onChange={(value) => (tab = value)}>
			<tabs.list>
				<tabs.trigger value="camera">Camera</tabs.trigger>
				<tabs.trigger value="document">Document</tabs.trigger>
			</tabs.list>
			<tabs.content value="camera">
				<CameraPanel />
			</tabs.content>
			<tabs.content value="document">
				<DocumentPanel />
			</tabs.content>
		</tabs.root>
	</main>
}
```

Confidence: 2/5

## Guesses

- The two elements' names (`CameraPreview` and `PdfPage`) and that they come from `@markless/device`.
- That they are capitalised components used as tags, not lowercase custom tags.
- The flip button: the sheet says a `base.button` without `pressed` has no state and doesn't say how it reports a press. So I made it a toggle button bound to `cam.facing === 'front'`, and its `onChange` calls `flip()`. If plain `base.button` accepted `onClick`, I would use that instead.
- That an async `onClick` handler that assigns state after an `await` is allowed.
- That `@if (photo)` narrows `photo` to `Photo` inside the block.
- That `pages ?? '…'` works as a plain expression in `{}`.
- That a `<style>` element with a template string is the accepted way to ship CSS from a page, and that `class` reaches the device element's box.
- That `camera()` is safe to call inside a panel that stays in the page while hidden, and that `ready` stays false until the panel is visible.
- That `tabs.root`'s `value` follows state when I pass it in and change it in `onChange`.
- That native `<button disabled>` and `onClick` work as in HTML.

## Website

Camera preview: after the browser's permission prompt, the visitor sees the browser's live webcam picture (a `<video>`-like box) filling the panel's width. If they deny access, they see an empty box.

PDF page: the visitor sees the browser's built-in PDF viewer showing the current page of `/guide.pdf`, with the browser's own viewer chrome, filling the panel's width.

## Answers

A: Both come from `@markless/device`, because that package already owns `camera()` and the `Camera` type, and the sheet says both boxes are drawn by the device (system preview and PDF view on phones, browser features on the web).

B: `<CameraPreview el={cam} />`. Two other spellings I would find natural: `<CameraView el={cam} />` and `<device.camera el={cam} />`.

C: Neither. The capitalised `CameraPreview` reads as an imported component, not an HTML element, because HTML tags are lowercase. It also isn't written as a `family.part` member like `tabs.trigger` or `base.button`, so a reader wouldn't take it for a `@markless/ui` part. A reader could still miss that it is a box drawn by the device that can't be styled.
