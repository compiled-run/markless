import type { Camera } from '@markless/device';

// Shows a camera made with camera(); fills its box with the live preview.
export declare function CameraView(props: { el: Camera; class?: string }): Element;

// Shows one page of a PDF; `page` counts from 1; onLoad is called once with the page count.
export declare function PdfView(props: {

```tsrx
// file: pages/media.tsrx
import { state } from '@markless/core';
import { camera, CameraView, PdfView } from '@markless/device';
import type { Photo } from '@markless/device';
import { tabs, base } from '@markless/ui';

function CameraPanel() @{
	const cam = camera({ facing: 'back' });
	let photo = state<Photo | null>(null);
	let busy = state(false);

	<section class="camera-panel">
		<CameraView el={cam} class="fill-width" />
		@if (cam.access === 'denied') {
			<p>Camera access was denied.</p>
		}
		<div class="controls">
			<button
				disabled={busy || cam.access === 'denied'}
				onClick={async () => {
					busy = true;
					try {
						photo = await cam.takePhoto();
					} finally {
						busy = false;
					}
				}}
			>
				Take photo
			</button>
			<base.button
				pressed={cam.facing === 'front'}
				disabled={!cam.ready}
				onChange={() => cam.flip()}
			>
				Flip camera
			</base.button>
		</div>
		@if (photo) {
			<img class="fill-width" src={photo.url} alt="Last photo taken" />
		}
	</section>
}

function DocumentPanel() @{
	let page = state(1);
	let pages = state(0);

	<section class="document-panel">
		<PdfView
			src="/guide.pdf"
			page={page}
			onLoad={(info) => (pages = info.pages)}
			class="fill-width"
		/>
		<div class="controls">
			<button disabled={page <= 1} onClick={() => page--}>Previous</button>
			<span>Page {page} of {pages}</span>
			<button disabled={pages === 0 || page >= pages} onClick={() => page++}>Next</button>
		</div>
	</section>
}

export default function MediaPage() @{
	let tab = state('camera');

	<main>
		<style>{`
			.fill-width { display: block; width: 100%; }
			.controls { display: flex; gap: 0.5rem; align-items: center; }
		`}</style>
		<tabs.root value={tab} onChange={(value) => (tab = value)}>
			<tabs.list>
				<tabs.trigger value="camera">Camera</tabs.trigger>
				<tabs.trigger value="document">Document</tabs.trigger>
			</tabs.list>
			<tabs.content value="camera">
				<CameraPanel />
			</tabs.content>
			<tabs.content value="document">
				<DocumentPanel />
			</tabs.content>
		</tabs.root>
	</main>
}
```

## Guesses

- The names `CameraView` and `PdfView`, and that they are imported from `@markless/device` as components.
- `base.button` has no `onClick` prop. So I made the flip button a toggle: its `pressed` is tied to `cam.facing === 'front'`, and `onChange` calls `flip()`. I assumed the controlled `pressed` value follows the camera's live `facing` field.
- The shutter, Previous and Next buttons are plain `<button>` elements with native `onClick`. I assumed an async handler that assigns state is allowed.
- `tabs.root` may be controlled with `value` and `onChange`. A plain `value="camera"` might also work, but I bound it to state to be safe.
- The camera keeps running while its tab is hidden, because a content panel stays in the page.
- I assumed a `<style>` block with a template string works inside a `.tsrx` root. Otherwise the classes would live in a global stylesheet.
- `@if` with no `@else`. I assumed `photo` narrows to non-null inside the block.
- I assumed `try/finally` inside a handler is fine.

## Website

Camera preview: the browser asks for camera permission, then the box shows the browser's live camera picture (like a `<video>` from getUserMedia), filling the panel width.

PDF page: the box shows `/guide.pdf` in the browser's built-in PDF viewer, with its own toolbar and styling, opened at the current page and filling the panel width.

## Answers

A: Both come from `@markless/device`. That package already owns `camera()` and the `Camera` type, and these elements are drawn by the device itself (system camera preview or PDF view) rather than being unstyled behaviour parts. That rules out `@markless/ui`, and they are not plain HTML.

B: I wrote `<CameraView el={cam} />`. Two other natural spellings are `<CameraPreview>` and `<device.camera>`.

C: A reader would not take it for HTML, because the capitalised name marks an imported component. They also would not mistake it for a `@markless/ui` part, because those use lowercase `family.part` dot access (`tabs.trigger`, `base.button`). It could, however, be confused with the `Camera` type from the same module.
