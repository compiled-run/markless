```ts
// Module: @markless/device (guessed additions)
import type { Camera } from '@markless/device';

// Shows a camera made with camera(); the device draws the preview inside the box.
export declare function CameraView(props: { el: Camera; class?: string }): Element;

// Shows one page of a PDF; `page` counts from 1; `onLoad` is called once with the page count.
export declare function PdfView(props: {
	src: string;
	page?: number;
	onLoad?: (info: { pages: number }) => void;
	class?: string;
}): Element;
```

```tsrx
// file: pages/media.tsrx
import { state } from '@markless/core';
import { camera, CameraView, PdfView } from '@markless/device';
import type { Photo } from '@markless/device';
import { base, tabs } from '@markless/ui';

function CameraPanel() @{
	const cam = camera({ facing: 'back' });
	let last = state<Photo | null>(null);

	<div class="camera-panel">
		<CameraView el={cam} class="fill-width" />
		<div class="camera-controls">
			<button
				disabled={!cam.ready}
				onClick={async () => (last = await cam.takePhoto())}
			>
				Take photo
			</button>
			<base.button
				pressed={cam.facing === 'front'}
				onChange={() => cam.flip()}
			>
				Flip camera
			</base.button>
		</div>
		@if (last) {
			<img class="fill-width" src={last.url} alt="Last photo taken" />
		}
	</div>
}

function DocumentPanel() @{
	let page = state(1);
	let pages = state(0);

	<div class="document-panel">
		<PdfView
			src="/guide.pdf"
			page={page}
			onLoad={(info) => (pages = info.pages)}
			class="fill-width"
		/>
		<div class="pdf-controls">
			<button disabled={page <= 1} onClick={() => page--}>Previous</button>
			<span>Page {page} of {pages}</span>
			<button disabled={pages === 0 || page >= pages} onClick={() => page++}>Next</button>
		</div>
	</div>
}

export default function Media() @{
	let tab = state('camera');

	<main>
		<style>
			{`.fill-width { display: block; width: 100%; }
			.camera-panel :where(.fill-width) { aspect-ratio: 3 / 4; }
			.document-panel :where(.fill-width) { aspect-ratio: 1 / 1.414; }
			[ui-selected] { font-weight: bold; }`}
		</style>
		<tabs.root value={tab} onChange={(value) => (tab = value)}>
			<tabs.list>
				<tabs.trigger value="camera">Camera</tabs.trigger>
				<tabs.trigger value="document">Document</tabs.trigger>
			</tabs.list>
			<tabs.content value="camera">
				<CameraPanel />
			</tabs.content>
			<tabs.content value="document">
				<DocumentPanel />
			</tabs.content>
		</tabs.root>
	</main>
}
```

## Guesses

- **Element names and import path:** the names `CameraView` and `PdfView`, and that both are imported from `@markless/device`. The sheet does not publish either.
- **Flip button:** `base.button` has no `onClick` prop. So I made it a toggle (`pressed` = front camera is active) and call `cam.flip()` from `onChange`. I am assuming this works even though `pressed` is controlled by `cam.facing`.
- **Shutter button:** that an `async` arrow is fine as an `onClick` handler, and that assigning the awaited result to `last` updates the view.
- **Showing the photo:** that `@if (last)` narrows `last` to a `Photo` inside the block.
- **Reading live fields:** that `cam.ready` and `cam.facing` can be read straight in markup and stay live, since the sheet calls the fields "live".
- **Styling:** that a plain `<style>` block with a template string is acceptable on a page. I could also have used an external stylesheet. I am also assuming the element boxes accept `display: block; width: 100%`.
- **Tab state:** that `tabs.root` is controlled by `value`/`onChange`, and that setting `value` to `'camera'` makes Camera show first.
- **Camera lifetime:** `tabs.content` keeps hidden panels in the page, so the camera stays held, and may keep running, while the Document tab is showing.
- **Page count before load:** `pages` reads 0 until `onLoad` runs, so the label briefly shows "Page 1 of 0".

## Website

- **Camera preview:** the browser's own live camera picture, after the visitor allows camera access in the browser's permission prompt. It is sized to the panel width.
- **PDF page:** the browser's built-in PDF viewer showing page 1 of `/guide.pdf`, with that viewer's own controls. It is sized to the panel width.

## Answers

A: I import both from `@markless/device`. That package already owns `camera()`, `Camera` and `Photo`. These elements are drawn by the device (the system preview and the system PDF view), so they belong with the device features and not with `@markless/ui`, which holds unstyled behaviour families. Using one import for the camera handle and its view also keeps them together.

B: `<CameraView el={cam} />`. Two other spellings I would find natural are `<CameraPreview el={cam} />` and `<device.camera el={cam} />`.

C: A reader could not take it for either. HTML tag names are lowercase, so the capitalised `CameraView` is clearly not HTML. It is also not written in the `family.part` dotted form (`tabs.trigger`, `base.button`) used by `@markless/ui` parts. It reads as an ordinary imported component, and its import from `@markless/device` places it.
