```tsrx
// file: pages/media.tsrx
import { state } from '@markless/core';
import { camera, CameraView, PdfView } from '@markless/device';
import type { Photo } from '@markless/device';
import { base, tabs } from '@markless/ui';

function CameraPanel() @{
	const cam = camera({ facing: 'back' });
	let last = state<Photo | null>(null);

	<div class="camera-panel">
		<CameraView el={cam} class="fill-width" />
		<div class="camera-controls">
			<button
				disabled={!cam.ready}
				onClick={async () => (last = await cam.takePhoto())}
			>
				Take photo
			</button>
			<base.button
				pressed={cam.facing === 'front'}
				onChange={() => cam.flip()}
			>
				Flip camera
			</base.button>
		</div>
		@if (last) {
			<img class="fill-width" src={last.url} width={last.width} height={last.height} alt="Last photo" />
		}
	</div>
}

function DocumentPanel() @{
	let page = state(1);
	let pages = state(0);

	<div class="document-panel">
		<PdfView
			src="/guide.pdf"
			page={page}
			onLoad={(info) => (pages = info.pages)}
			class="fill-width"
		/>
		<p>Page {page} of {pages}</p>
		<div class="pdf-controls">
			<button disabled={page <= 1} onClick={() => page--}>Previous</button>
			<button disabled={page >= pages} onClick={() => page++}>Next</button>
		</div>
	</div>
}

export default function Media() @{
	let tab = state('camera');

	<main>
		<style>{`
			.fill-width { display: block; width: 100%; }
			.camera-panel .fill-width { aspect-ratio: 3 / 4; height: auto; }
			.document-panel .fill-width { aspect-ratio: 1 / 1.414; height: auto; }
		`}</style>
		<tabs.root value={tab} onChange={(value) => (tab = value)}>
			<tabs.list>
				<tabs.trigger value="camera">Camera</tabs.trigger>
				<tabs.trigger value="document">Document</tabs.trigger>
			</tabs.list>
			<tabs.content value="camera">
				<CameraPanel />
			</tabs.content>
			<tabs.content value="document">
				<DocumentPanel />
			</tabs.content>
		</tabs.root>
	</main>
}
```

## Guesses

- The element names `CameraView` and `PdfView`, and that both are imported from `@markless/device`.
- That they are used as capitalised components and take exactly the props on the sheet.
- `base.button` has no `onClick`. So the flip button is a toggle button: `pressed` follows `cam.facing === 'front'`, and `onChange` calls `cam.flip()`. I guessed that a `pressed` value that is passed in follows the camera's live state instead of keeping its own copy.
- The shutter, Previous and Next buttons are plain HTML `<button>`s, because `base.button` gives no way to react to a plain press.
- That a `<style>` block inside the page works, and that the device elements accept sizing from `class`.
- That `async` click handlers that assign state after an `await` are allowed.
- That `pages` reads 0 until `onLoad` fires, so the label briefly says "Page 1 of 0".
- That a camera inside a tab panel that is not showing stays running, since panels stay in the page.

## Website

Camera preview: the browser's own live camera picture, shown after the browser asks for camera permission, filling the panel width.

PDF page: the browser's built-in PDF viewer showing the chosen page of `/guide.pdf`, with the browser's own look, filling the panel width.

## Answers

A: From `@markless/device`. `camera()` and the `Camera` type are already there, and these elements only exist to show device-drawn content (the camera picture and the PDF view), so I expect them next to the functions that feed them, not in `@markless/ui`, whose parts carry behaviour and `ui-*` styling hooks.

B: `<CameraView el={cam} />`. Two other spellings I would find natural are `<CameraPreview el={cam} />` and `<device.camera el={cam} />`.

C: Neither. `CameraView` is capitalised, so a reader takes it for an imported component, not an HTML tag. It is also not written as `family.part` the way `tabs.root` and `base.button` are, so it does not look like a `@markless/ui` part. A reader could still wonder whether it is plain user code, and only the import line from `@markless/device` answers that.
