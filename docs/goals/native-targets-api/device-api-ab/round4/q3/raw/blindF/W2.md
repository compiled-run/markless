```tsx
// file: pages/media.tsrx
import { state } from '@markless/core';
import { camera, pdf, View, type Photo } from '@markless/device';
import { tabs, base } from '@markless/ui';

function CameraPanel() @{
	const cam = camera({ facing: 'back' });
	let last = state<Photo | null>(null);

	<div class="panel">
		<View el={cam} class="fill" />
		<div class="controls">
			<button disabled={!cam.ready} onClick={async () => (last = await cam.takePhoto())}>Take photo</button>
			<base.button pressed={cam.facing === 'front'} onChange={() => cam.flip()}>
				Flip camera
			</base.button>
		</div>
		@if (last) {
			<img class="fill" src={last.url} width={last.width} height={last.height} alt="Last photo" />
		}
	</div>
}

function DocumentPanel() @{
	const doc = pdf('/guide.pdf');

	<div class="panel">
		<View el={doc} class="fill" />
		<div class="controls">
			<button disabled={doc.page <= 1} onClick={() => (doc.page = doc.page - 1)}>Previous</button>
			@if (doc.pages > 0) {
				<span>Page {doc.page} of {doc.pages}</span>
			} @else {
				<span>Loading…</span>
			}
			<button disabled={doc.pages === 0 || doc.page >= doc.pages} onClick={() => (doc.page = doc.page + 1)}>
				Next
			</button>
		</div>
	</div>
}

export default function Media() @{
	let tab = state('camera');

	<main>
		<style>{`
			.panel { display: flex; flex-direction: column; gap: 0.5rem; }
			.fill { display: block; width: 100%; }
			.controls { display: flex; gap: 0.5rem; align-items: center; }
		`}</style>
		<tabs.root value={tab} onChange={(value) => (tab = value)}>
			<tabs.list>
				<tabs.trigger value="camera">Camera</tabs.trigger>
				<tabs.trigger value="document">Document</tabs.trigger>
			</tabs.list>
			<tabs.content value="camera">
				<CameraPanel />
			</tabs.content>
			<tabs.content value="document">
				<DocumentPanel />
			</tabs.content>
		</tabs.root>
	</main>
}
```

## Guesses

- The element's name (`View`), that it is a capitalised component and that it is exported from `@markless/device`. The sheet does not publish any of these.
- That `Photo` can be imported as a type from `@markless/device`.
- `base.button` lists no `onClick`. So the flip button is a toggle: `pressed` follows whether the front camera is on, and `onChange` calls `cam.flip()`. I am assuming `onChange` fires on each press and that `pressed` stays in sync with `cam.facing`.
- That assigning a field on a live device object (`doc.page = ...`) updates the page the same way a `state` assignment does.
- That a camera made in a tab that is not showing still works, since the tab's content stays in the page.
- That `<style>` with a template string is allowed in a page. Plain class names in a separate stylesheet would also work.
- That `@if` / `@else` is written this way and that the shutter's async handler is allowed.

## Answers

A: From `@markless/device`. That package already makes the objects it shows (`camera()`, `pdf()`), one element is used for every kind of device object, and only the device layer can draw the native preview or PDF view. It is not a `@markless/ui` part, because the sheet says it has no parts and no behaviour.

B: `<View el={cam} />`. Two other spellings I would also find natural: `<device.view el={cam} />` and `<Surface el={cam} />`.
