// Module: @markless/device
// One element shows any device object (Camera, Pdf, ...). The device draws inside it; the author only sizes and places the box.
export declare function DeviceView(props: { el: Camera | Pdf; class?: string }): Element;

```tsrx
// file: pages/media.tsrx
import { state } from '@markless/core';
import { camera, pdf, DeviceView, type Photo } from '@markless/device';
import { tabs, base } from '@markless/ui';

function CameraPanel() @{
	const cam = camera({ facing: 'back' });
	let last = state<Photo | null>(null);

	<div class="panel">
		<DeviceView el={cam} class="fill" />
		<div class="controls">
			<button onClick={async () => (last = await cam.takePhoto())}>Take photo</button>
			<base.button pressed={cam.facing === 'front'} onChange={() => cam.flip()}>
				Flip camera
			</base.button>
		</div>
		@if (last) {
			<img class="fill" src={last.url} alt="Last photo taken" />
		}
	</div>
}

function DocumentPanel() @{
	const doc = pdf('/guide.pdf');

	<div class="panel">
		<DeviceView el={doc} class="fill" />
		<div class="controls">
			<button disabled={doc.page <= 1} onClick={() => doc.page--}>Previous</button>
			<span>Page {doc.page} of {doc.pages}</span>
			<button disabled={doc.pages === 0 || doc.page >= doc.pages} onClick={() => doc.page++}>Next</button>
		</div>
	</div>
}

export default function Media() @{
	let tab = state('camera');

	<main>
		<style>{`
			.panel { display: flex; flex-direction: column; gap: 0.5rem; }
			.fill { display: block; width: 100%; }
			.controls { display: flex; gap: 0.5rem; align-items: center; }
		`}</style>
		<tabs.root value={tab} onChange={(value) => (tab = value)}>
			<tabs.list>
				<tabs.trigger value="camera">Camera</tabs.trigger>
				<tabs.trigger value="document">Document</tabs.trigger>
			</tabs.list>
			<tabs.content value="camera">
				<CameraPanel />
			</tabs.content>
			<tabs.content value="document">
				<DocumentPanel />
			</tabs.content>
		</tabs.root>
	</main>
}
```

## Guesses

The element's name and where it comes from: I guessed `DeviceView`, imported from `@markless/device`, and used as a capitalised component.

`base.button` has no `onClick`. I made the flip button a toggle, with `pressed` tied to `cam.facing === 'front'` and `onChange` calling `cam.flip()`. I also assumed a toggle whose `pressed` comes from the camera stays in step with the camera.

The shutter is a native `<button>` with an async `onClick` handler. I assumed assigning `last` after the `await` updates the page.

I assumed `@if (last)` narrows the type, so `last.url` type-checks inside the block.

I assumed `doc.page--` and `doc.page++` work, because the sheet says to assign to `page`. I also assumed the PDF fields update the page as they change.

I assumed a `<style>` block inside the component is allowed, and that a template string inside it is fine. A separate stylesheet would also work.

I assumed `tabs.root` is controlled through `value` and `onChange`, and that `state('camera')` gives a `string`, which matches `onChange`.

I assumed the camera starts while its tab is hidden, because hidden panels stay in the page. Hidden panels may still count as "on screen", so `ready` might behave in ways I can't see.

## Website

Camera preview: the browser asks for camera permission, then shows the live webcam picture as wide as the panel. Most desktops have only one camera, so flipping may do nothing visible.

PDF page: the browser's own PDF viewer shows `/guide.pdf` as wide as the panel, opened at the current page, with that viewer's own toolbar.

## Answers

A: It comes from `@markless/device`. The element exists only to show objects from that package (`camera()`, `pdf()`), and it gets its own props from those objects. So it should sit next to the functions that make them, not in `@markless/ui`, whose families have parts and behaviour.

B: `DeviceView`. Two other spellings I would find natural: `<device.view el={...} />` (a namespace object like `tabs.root`) and `<Surface el={...} />`.
