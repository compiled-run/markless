// Two elements have no HTML equivalent. Their tag names and where they come from are not published.
// Each is one box with no parts and no behaviour of its own. The author sizes and places the box with `class`; what is
// inside it is drawn by the device and cannot be styled: on iPhone and Android, the system's own camera preview and PDF
// view; on the website, the browser's camera picture and the browser's PDF viewer.
// The camera element shows a camera made with camera(). Its props:
interface CameraElementProps { el: Camera; class?: string }
// The PDF element shows one page of a PDF. `page` counts from 1; `onLoad` is called once with the page count. Its props:
interface PdfElementProps { src: string; page?: number; onLoad?: (info: { pages: number }) => void; class?: string }
