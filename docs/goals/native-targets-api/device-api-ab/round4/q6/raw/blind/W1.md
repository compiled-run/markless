import { state } from '@markless/core';
import { modal } from '@markless/ui';

type Address = { id: number; label: string };

const addresses: readonly Address[] = [

	{ id: 2, label: 'Work — 400 Market Ave, Floor 3' },
	{ id: 3, label: 'Parents — 7 Lakeview Road' },

export default function Checkout() @{
	let filtersOpen = state(false);
	let vegan = state(false);
	let glutenFree = state(false);
	let underTen = state(false);

	let deleteOpen = state(false);
	let deleted = state(false);

	let addressOpen = state(false);
	let chosen = state<Address>(addresses[0]);

	<main class="checkout">
		<section class="filters">
			<modal.root adaptive open={filtersOpen} onChange={(open) => (filtersOpen = open)}>
				<modal.trigger class="button">Filters</modal.trigger>
				<modal.backdrop class="backdrop">
					<modal.content class="panel">
						<modal.title>Filters</modal.title>
						<label>
							<input type="checkbox" checked={vegan} onChange={(event) => (vegan = event.currentTarget.checked)} />
							Vegan
						</label>
						<label>
							<input type="checkbox" checked={glutenFree} onChange={(event) => (glutenFree = event.currentTarget.checked)} />
							Gluten-free
						</label>
						<label>
							<input type="checkbox" checked={underTen} onChange={(event) => (underTen = event.currentTarget.checked)} />
							Under $10
						</label>
						<modal.close class="button primary">Apply</modal.close>
					</modal.content>
				</modal.backdrop>
			</modal.root>
		</section>

		<section class="order">
			@if (deleted) {
				<p>Order deleted.</p>
			} @else {
				<modal.root alert open={deleteOpen} onChange={(open) => (deleteOpen = open)}>
					<modal.trigger class="button danger">Delete</modal.trigger>
					<modal.backdrop class="backdrop">
						<modal.content class="dialog">
							<modal.title>Delete this order?</modal.title>
							<div class="actions">
								<modal.close class="button">Cancel</modal.close>
								<button
									class="button danger"
									onClick={() => {
										deleted = true;
										deleteOpen = false;
									}}
								>
									Delete
								</button>
							</div>
						</modal.content>
					</modal.backdrop>
				</modal.root>
			}
		</section>

		<section class="deliver">
			<p>

								</li>
							}

				</modal.backdrop>

		</section>
	</main>

## Phone
1. Filters: bottom sheet (an adaptive modal)
2. Delete: centered box (an alert modal)
3. Deliver to / Change: bottom sheet (an adaptive modal)

## Website
1. Filters: centered box
2. Delete: centered box
3. Deliver to / Change: centered box

## Guesses
- The opt-in spelling, `adaptive`, as a boolean prop on `modal.root`, and whether it also goes on a plain `<dialog>`.
- That `modal.close` closes the modal and fires `onChange(false)` while the parent controls `open`.
- That `modal.trigger` opens a modal whose `open` is controlled without an explicit `onClick`.
- That a plain `<button>` inside `modal.content` may close the modal by setting the controlled `open` state to false.
- That `checked`/`onChange` and `aria-pressed` pass through to native elements as usual.

## Answers
A: I used `adaptive` (`<modal.root adaptive>`). Two other spellings I would find natural: `sheet="native"` and `presentation="adaptive"` (or `mobileSheet`).
B: I used it on the Filters panel and the address picker. On a phone, both are pick-from-options panels that people expect to rise from the bottom within thumb reach. On a laptop a centered box fits better than a sheet stuck to the bottom of a wide screen. I left it off the Delete confirmation: that is an alert, and phones show alerts as centered boxes too, so it stays centered everywhere.
