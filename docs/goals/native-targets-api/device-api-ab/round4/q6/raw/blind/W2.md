```tsrx
// file: pages/checkout.tsrx
import { state } from '@markless/core';
import { drawer, modal } from '@markless/ui';

const addresses = [
	{ id: 1, label: 'Home, 12 Oak Street' },
	{ id: 2, label: 'Work, 400 Market Avenue' },
	{ id: 3, label: 'Mum, 7 Hillside Road' },
];

export default function Checkout() @{
	let filtersOpen = state(false);
	let vegan = state(false);
	let glutenFree = state(false);
	let underTen = state(false);

	let deleteOpen = state(false);
	let deleted = state(false);

	let addressOpen = state(false);
	let chosen = state(addresses[0].label);

	<main>
		<drawer.root open={filtersOpen} onChange={(open) => (filtersOpen = open)}>
			<drawer.trigger>Filters</drawer.trigger>
			<drawer.backdrop>
				<drawer.content>
					<drawer.title>Filters</drawer.title>
					<label>
						<input type="checkbox" checked={vegan} onChange={(event) => (vegan = event.currentTarget.checked)} />
						Vegan
					</label>
					<label>
						<input type="checkbox" checked={glutenFree} onChange={(event) => (glutenFree = event.currentTarget.checked)} />
						Gluten-free
					</label>
					<label>
						<input type="checkbox" checked={underTen} onChange={(event) => (underTen = event.currentTarget.checked)} />
						Under $10
					</label>
					<drawer.close>Apply</drawer.close>
				</drawer.content>
			</drawer.backdrop>
		</drawer.root>

		<modal.root alert open={deleteOpen} onChange={(open) => (deleteOpen = open)}>
			<modal.trigger>Delete</modal.trigger>
			<modal.backdrop>
				<modal.content>
					<modal.title>Delete this order?</modal.title>
					<modal.close onClick={() => (deleted = true)}>Delete</modal.close>
					<modal.close>Cancel</modal.close>
				</modal.content>
			</modal.backdrop>
		</modal.root>

		<p>
			Deliver to: {chosen}
			<modal.root adaptive open={addressOpen} onChange={(open) => (addressOpen = open)}>
				<modal.trigger>Change</modal.trigger>
				<modal.backdrop>
					<modal.content>
						<modal.title>Saved addresses</modal.title>
						<ul>
							@for (const address of addresses; key address.id) {
								<li>
									<button
										onClick={() => {
											chosen = address.label;
											addressOpen = false;
										}}
									>
										{address.label}
									</button>
								</li>
							}
						</ul>
						<modal.close>Cancel</modal.close>
					</modal.content>
				</modal.backdrop>
			</modal.root>
		</p>
	</main>
}
```

## Phone
1. Filters: bottom sheet (`drawer`)
2. Delete: centered box (alert `modal`)
3. Change address: bottom sheet (`modal` with `adaptive`)

## Website
1. Filters: bottom sheet (`drawer`)
2. Delete: centered box
3. Change address: centered box

## Guesses
- The opt-in spelling `adaptive` as a boolean prop on `modal.root`.
- That `modal.close` and `drawer.close` close the dialog on their own, and that an `onClick` on `modal.close` runs before it closes.
- That setting `addressOpen = false` through the controlled `open` prop closes the modal.
- That `@for` works over a plain array that is not state.
- That `drawer.trigger` and `modal.trigger` render a button.
- That the `deleted` flag is enough to stand in for the real delete action.

## Answers
A: I used `adaptive` (`<modal.root adaptive>`). Two other spellings I would find natural: `sheet="auto"` (or `presentation="adaptive"`), and `adaptToPlatform`.

B: I used it on the "Deliver to" address picker. Picking from a short list is the classic phone bottom-sheet pattern: it sits within thumb reach and feels native. On a laptop, a centered box suits it better. I left the Delete confirmation as a centered alert on every platform, because a destructive yes/no question should be a focused alert dialog. I made Filters a `drawer`, so it is a bottom sheet everywhere, since the sheet says a drawer does that on every platform. I did not want to use the single opt-in on it.
