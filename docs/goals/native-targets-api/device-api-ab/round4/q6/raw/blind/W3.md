// file: pages/checkout.tsrx
import { state } from '@markless/core';
import { modal } from '@markless/ui';

const addresses = [
	{ id: 1, label: 'Home, 12 Elm Street' },
	{ id: 2, label: 'Work, 400 Market Street' },
	{ id: 3, label: "Mum's, 7 Harbour Road" },
];

export default function Checkout() @{
	let filtersOpen = state(false);
	let vegan = state(false);
	let glutenFree = state(false);
	let underTen = state(false);

	let deleteOpen = state(false);
	let deleted = state(false);

	let addressOpen = state(false);
	let address = state(addresses[0].label);

	<main>
		<modal.root open={filtersOpen} onChange={(open) => (filtersOpen = open)} adaptive>
			<modal.trigger>Filters</modal.trigger>
			<modal.backdrop>
				<modal.content>
					<modal.title>Filters</modal.title>
					<label>
						<input type="checkbox" checked={vegan} onChange={(event) => (vegan = event.currentTarget.checked)} />
						Vegan
					</label>
					<label>
						<input type="checkbox" checked={glutenFree} onChange={(event) => (glutenFree = event.currentTarget.checked)} />
						Gluten-free
					</label>
					<label>
						<input type="checkbox" checked={underTen} onChange={(event) => (underTen = event.currentTarget.checked)} />
						Under $10
					</label>
					<modal.close>Apply</modal.close>
				</modal.content>
			</modal.backdrop>
		</modal.root>

		<modal.root alert open={deleteOpen} onChange={(open) => (deleteOpen = open)}>
			<modal.trigger>Delete</modal.trigger>
			<modal.backdrop>
				<modal.content>
					<modal.title>Delete this order?</modal.title>
					<modal.close onClick={() => (deleted = true)}>Delete</modal.close>
					<modal.close>Cancel</modal.close>
				</modal.content>
			</modal.backdrop>
		</modal.root>
		@if (deleted) {
			<p>Order deleted.</p>
		}

		<p>
			Deliver to: {address}
			<modal.root open={addressOpen} onChange={(open) => (addressOpen = open)} adaptive>
				<modal.trigger>Change</modal.trigger>
				<modal.backdrop>
					<modal.content>
						<modal.title>Saved addresses</modal.title>
						<ul>
							@for (const item of addresses; key item.id) {
								<li>
									<button
										onClick={() => {
											address = item.label;
											addressOpen = false;
										}}
									>
										{item.label}
									</button>
								</li>
							}
						</ul>
					</modal.content>
				</modal.backdrop>
			</modal.root>
		</p>
	</main>
}

## Guesses
- The opt-in spelling: a boolean `adaptive` prop on `modal.root`.
- That `modal.trigger` opens the modal and `modal.close` closes it on its own, and that both call `onChange` so the controlled `open` state stays correct (the sheet says this for `drawer` but not for `modal`).
- That a `modal.close` with an `onClick` runs the handler and then closes, which is how the Delete button records the deletion.
- That closing the address list by setting `addressOpen = false` from a plain button works when `open` is controlled.
- That a modal root may sit inside a `<p>` next to text, and that the `alert` and `adaptive` props can be used together (here they are not).

## Answers
A: I used `adaptive` (`<modal.root adaptive>`). Two other spellings I would find natural: `sheet="native"` / `presentation="adaptive"`, or `platformSheet`.
B: I used it on the Filters panel and the address picker. Both are pick-from-a-list panels that phone users expect to rise from the bottom within thumb reach, and on a laptop a centered box works fine for them. I left it off the Delete confirmation because a destructive alert should stay a centered alert box on every platform, as iOS and Android alert dialogs are.
