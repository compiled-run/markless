// file: pages/checkout.tsrx
import { state } from '@markless/core';
import { modal } from '@markless/ui';

type Address = { id: number; label: string };

const addresses: readonly Address[] = [
	{ id: 1, label: 'Home, 12 Elm Street' },
	{ id: 2, label: 'Work, 400 Market Street' },
	{ id: 3, label: "Mum's, 7 Harbour Road" },
];

export default function Checkout() @{
	let filtersOpen = state(false);
	let vegan = state(false);
	let glutenFree = state(false);
	let underTen = state(false);

	let deleteOpen = state(false);
	let deleted = state(false);

	let addressOpen = state(false);
	let address = state<Address>(addresses[0]);

	<main class="checkout">
		<modal.root adaptive open={filtersOpen} onChange={(open) => (filtersOpen = open)}>
			<modal.trigger class="filters-button">Filters</modal.trigger>
			<modal.backdrop class="backdrop">
				<modal.content class="panel">
					<modal.title>Filters</modal.title>
					<fieldset>
						<label>
							<input type="checkbox" checked={vegan} onChange={(event) => (vegan = event.currentTarget.checked)} />
							Vegan
						</label>
						<label>
							<input type="checkbox" checked={glutenFree} onChange={(event) => (glutenFree = event.currentTarget.checked)} />
							Gluten-free
						</label>
						<label>
							<input type="checkbox" checked={underTen} onChange={(event) => (underTen = event.currentTarget.checked)} />
							Under $10
						</label>
					</fieldset>
					<modal.close class="apply">Apply</modal.close>
				</modal.content>
			</modal.backdrop>
		</modal.root>

		<modal.root alert open={deleteOpen} onChange={(open) => (deleteOpen = open)}>
			<modal.trigger class="delete-button">Delete</modal.trigger>
			<modal.backdrop class="backdrop">
				<modal.content class="confirm">
					<modal.title>Delete this order?</modal.title>
					<div class="actions">
						<modal.close class="danger" onClick={() => (deleted = true)}>Delete</modal.close>
						<modal.close>Cancel</modal.close>
					</div>
				</modal.content>
			</modal.backdrop>
		</modal.root>

		<p class="deliver-to">
			Deliver to: <strong>{address.label}</strong>
			<modal.root adaptive open={addressOpen} onChange={(open) => (addressOpen = open)}>
				<modal.trigger class="change-button">Change</modal.trigger>
				<modal.backdrop class="backdrop">
					<modal.content class="panel">
						<modal.title>Saved addresses</modal.title>
						<ul>
							@for (const option of addresses; key option.id) {
								<li>
									<button
										aria-pressed={option.id === address.id}
										onClick={() => {
											address = option;
											addressOpen = false;
										}}
									>
										{option.label}
									</button>
								</li>
							}
						</ul>
					</modal.content>
				</modal.backdrop>
			</modal.root>
		</p>
	</main>
}

## Phone
1. Filters: bottom sheet, because the modal opts in to adapting.
2. Delete: centered box, an alert dialog.
3. Deliver to / Change: bottom sheet, because the modal opts in to adapting.

## Website
1. Filters: centered box.
2. Delete: centered box, an alert dialog.
3. Deliver to / Change: centered box.

## Guesses
- The opt-in is a bare boolean prop `adaptive` on `modal.root`.
- `modal.close` takes an extra `onClick` and still closes the dialog, so the Delete button can record the deletion and close in one press.
- Setting the controlled `open` state to `false` from a handler inside the content closes the address list.
- Native `<input type="checkbox">` with `checked` and `onChange` binds to state as in the browser, and `event.currentTarget.checked` is typed.
- `aria-pressed` on a plain `<button>` is enough to show which address is chosen.
- It is fine to put a `modal.root` inside a `<p>`, as long as its rendered elements are not invalid inside a paragraph. If they are, the line should be a `<div>`.

## Answers
A: I used `adaptive` (a boolean on `modal.root`). Two other spellings I would find natural: `sheet="native"` and `presentation="adaptive"`.

B: I used it on the Filters panel and the address list. On a phone, both are short pick-and-dismiss tasks that are easiest to reach with a thumb from a bottom sheet. On a laptop, a centered box is the usual pattern. I did not use it on the Delete confirmation. A destructive yes/no question should stay a centered alert on every platform, because that interrupts the user and draws their attention. I picked the adaptive modal over `drawer` because `drawer` would also be a bottom sheet on the website.

Tools used: Read (only to open the task file, as instructed). No other tools.
