```tsrx
// file: pages/checkout.tsrx
import { state } from '@markless/core';
import { modal } from '@markless/ui';

const addresses = [
	{ id: 1, label: 'Home, 12 Elm Street' },
	{ id: 2, label: 'Work, 400 Market Avenue' },
	{ id: 3, label: 'Mum, 7 Harbour Road' },
];

export default function Checkout() @{
	let vegan = state(false);
	let glutenFree = state(false);
	let underTen = state(false);
	let address = state(addresses[0].label);
	let deleted = state(false);

	<main>
		<section>
			<modal.root adaptive>
				<modal.trigger>Filters</modal.trigger>
				<modal.backdrop>
					<modal.content>
						<modal.title>Filters</modal.title>
						<label>
							<input type="checkbox" checked={vegan} onChange={(event) => (vegan = event.currentTarget.checked)} />
							Vegan
						</label>
						<label>
							<input type="checkbox" checked={glutenFree} onChange={(event) => (glutenFree = event.currentTarget.checked)} />
							Gluten-free
						</label>
						<label>
							<input type="checkbox" checked={underTen} onChange={(event) => (underTen = event.currentTarget.checked)} />
							Under $10
						</label>
						<modal.close>Apply</modal.close>
					</modal.content>
				</modal.backdrop>
			</modal.root>
		</section>

		<section>
			<modal.root alert>
				<modal.trigger>Delete</modal.trigger>
				<modal.backdrop>
					<modal.content>
						<modal.title>Delete this order?</modal.title>
						<modal.close onClick={() => (deleted = true)}>Delete</modal.close>
						<modal.close>Cancel</modal.close>
					</modal.content>
				</modal.backdrop>
			</modal.root>
			@if (deleted) {
				<p>Order deleted.</p>
			}
		</section>

		<section>
			<p>
				Deliver to: {address}
			</p>
			<modal.root adaptive>
				<modal.trigger>Change</modal.trigger>
				<modal.backdrop>
					<modal.content>
						<modal.title>Saved addresses</modal.title>
						<ul>
							@for (const item of addresses; key item.id) {
								<li>
									<modal.close onClick={() => (address = item.label)}>{item.label}</modal.close>
								</li>
							}
						</ul>
					</modal.content>
				</modal.backdrop>
			</modal.root>
		</section>
	</main>
}
```

## Phone
1. Filters: bottom sheet.
2. Delete: centered box.
3. Change address: bottom sheet.

## Website
1. Filters: centered box.
2. Delete: centered box.
3. Change address: centered box.

## Guesses
- The opt-in is a boolean `adaptive` prop on `modal.root`.
- `modal.close` closes the dialog and still runs its `onClick`, so Apply and each address button can be close buttons.
- `modal.trigger` opens an uncontrolled `modal.root` when `open` is not given, the same way `drawer.trigger` does.
- Checkbox `checked` binding with `onChange` and `event.currentTarget.checked` works like native DOM.
- A module-level constant array can be used in `@for`.
- `@if` without `@else` is allowed.

## Answers
A: I used `<modal.root adaptive>`. Two other spellings I would find natural are `<modal.root sheet="native">` and `<modal.root presentation="adaptive">`.

B: I used it on the Filters panel and the saved-address list. On a phone both are pick-from-a-list tasks that people expect in a bottom sheet within reach of the thumb. On a laptop a centered box fits better than a panel sliding up from the bottom of a wide screen. I did not use it on the Delete confirmation. That is an alert dialog, and a centered box that stops the user is the normal look for a destructive confirmation on every platform.
